# Contact Center Resource Portal — Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Build a Microsoft Dynamics 365-styled Azure Static Web App knowledge portal that serves documents, videos, and images from Azure Blob Storage with metadata indexed in a GitHub-hosted `content-index.json` manifest.

**Architecture:** React (Vite) SPA + Azure Functions API routes deployed as a single Azure Static Web App unit. Files stored in private Azure Blob Storage served via short-lived SAS URLs. Manifest lives in GitHub repo, auto-regenerated by GitHub Action on every upload commit. Auth via SWA built-in Entra ID — zero custom auth code.

**Tech Stack:** React 18, TypeScript, Vite, React Router v6, Azure Functions (Node.js v4), `@azure/storage-blob`, `@octokit/rest`, Vitest, @testing-library/react, CSS custom properties (no CSS framework)

```typescript
import type { Resource, ResourceType, SortOrder } from '../types'

export function searchResources(resources: Resource[], query: string): Resource[] {
  if (!query.trim()) return resources
  const q = query.toLowerCase()
  return resources.filter(r =>
    r.title.toLowerCase().includes(q) ||
    r.description.toLowerCase().includes(q) ||
    r.category.toLowerCase().includes(q)
  )
}

export function filterByType(resources: Resource[], type: ResourceType | null): Resource[] {
  if (!type) return resources
  return resources.filter(r => r.type === type)
}

export function filterByCategory(resources: Resource[], category: string | null): Resource[] {
  if (!category) return resources
  return resources.filter(r => r.category === category)
}

export function sortResources(resources: Resource[], order: SortOrder): Resource[] {
  return [...resources].sort((a, b) => {
    const diff = new Date(b.uploadDate).getTime() - new Date(a.uploadDate).getTime()
    return order === 'newest' ? diff : -diff
  })
}
```

```typescript
import { createContext, useContext, useReducer, useEffect, ReactNode } from 'react'
import type { ContentManifest } from '../types'

interface ManifestState {
  manifest: ContentManifest | null
  loading: boolean
  error: string | null
}

type ManifestAction =
  | { type: 'LOADING' }
  | { type: 'LOADED'; payload: ContentManifest }
  | { type: 'ERROR'; payload: string }

function reducer(state: ManifestState, action: ManifestAction): ManifestState {
  switch (action.type) {
    case 'LOADING': return { ...state, loading: true, error: null }
    case 'LOADED': return { manifest: action.payload, loading: false, error: null }
    case 'ERROR': return { ...state, loading: false, error: action.payload }
  }
}

const ManifestContext = createContext<{
  state: ManifestState
  refresh: () => void
} | null>(null)

export function ManifestProvider({ children }: { children: ReactNode }) {
  const [state, dispatch] = useReducer(reducer, { manifest: null, loading: true, error: null })

  async function load() {
    dispatch({ type: 'LOADING' })
    try {
      const res = await fetch('/api/manifest')
      if (!res.ok) throw new Error(`HTTP ${res.status}`)
      const data: ContentManifest = await res.json()
      dispatch({ type: 'LOADED', payload: data })
    } catch (e) {
      dispatch({ type: 'ERROR', payload: (e as Error).message })
    }
  }

  useEffect(() => { load() }, [])

  return (
    <ManifestContext.Provider value={{ state, refresh: load }}>
      {children}
    </ManifestContext.Provider>
  )
}

export function useManifest() {
  const ctx = useContext(ManifestContext)
  if (!ctx) throw new Error('useManifest must be used inside ManifestProvider')
  return ctx
}
```

**Step 2: Create `src/context/AuthContext.tsx`**

```typescript
import { createContext, useContext, useEffect, useState, ReactNode } from 'react'
import type { AuthUser } from '../types'

interface AuthState {
  user: AuthUser | null
  loading: boolean
}

const AuthContext = createContext<AuthState>({ user: null, loading: true })

export function AuthProvider({ children }: { children: ReactNode }) {
  const [state, setState] = useState<AuthState>({ user: null, loading: true })

  useEffect(() => {
    fetch('/.auth/me')
      .then(r => r.json())
      .then(data => {
        const principal = data?.clientPrincipal
        if (principal) {
          setState({
            user: { email: principal.userDetails, roles: principal.userRoles ?? [] },
            loading: false,
          })
        } else {
          setState({ user: null, loading: false })
        }
      })
      .catch(() => setState({ user: null, loading: false }))
  }, [])

  return <AuthContext.Provider value={state}>{children}</AuthContext.Provider>
}

export function useAuth() {
  return useContext(AuthContext)
}
```

**Step 3: Create `src/hooks/useDebounce.ts`**

```typescript
import { useState, useEffect } from 'react'

export function useDebounce<T>(value: T, delay = 200): T {
  const [debounced, setDebounced] = useState(value)
  useEffect(() => {
    const id = setTimeout(() => setDebounced(value), delay)
    return () => clearTimeout(id)
  }, [value, delay])
  return debounced
}
```

```typescript
import { createContext, useContext, useState, ReactNode, useCallback } from 'react'
import './Toast.css'

type ToastType = 'success' | 'error' | 'info'
interface ToastItem { id: number; message: string; type: ToastType }

const ToastContext = createContext<(message: string, type?: ToastType) => void>(() => {})

let _id = 0

export function ToastProvider({ children }: { children: ReactNode }) {
  const [toasts, setToasts] = useState<ToastItem[]>([])

  const show = useCallback((message: string, type: ToastType = 'info') => {
    const id = ++_id
    setToasts(t => [...t, { id, message, type }])
    setTimeout(() => setToasts(t => t.filter(x => x.id !== id)), 3500)
  }, [])

  return (
    <ToastContext.Provider value={show}>
      {children}
      <div className="toast-container">
        {toasts.map(t => (
          <div key={t.id} className={`toast toast--${t.type}`}>{t.message}</div>
        ))}
      </div>
    </ToastContext.Provider>
  )
}

export function useToast() { return useContext(ToastContext) }
```

```typescript
    expiresOn: expiry,
```

```typescript
const MANIFEST_PATH = 'content-index.json'

export async function readManifest(): Promise<{ manifest: ContentManifest; sha: string }> {
  const octokit = getOctokit()
  const { data } = await octokit.repos.getContent({ owner: OWNER, repo: REPO, path: MANIFEST_PATH })
  if (Array.isArray(data) || data.type !== 'file') throw new Error('Manifest not found')
  const content = Buffer.from(data.content, 'base64').toString('utf8')
```

```typescript
import { app, HttpRequest, HttpResponseInit, InvocationContext } from '@azure/functions'
import { v4 as uuidv4 } from 'uuid'
import { uploadBlob } from '../shared/blobClient'
import { readManifest, writeManifest } from '../shared/githubClient'
import { requireContentManager, getCallerEmail } from '../shared/auth'
import type { Resource, ResourceType } from '../types'

const ALLOWED_TYPES: Record<string, ResourceType> = {
  pdf: 'document', docx: 'document', pptx: 'document', xlsx: 'document',
  mp4: 'video', webm: 'video',
  jpg: 'image', jpeg: 'image', png: 'image', gif: 'image', svg: 'image',
}

const MAX_SIZE = 100 * 1024 * 1024

async function handleUpload(request: HttpRequest, context: InvocationContext): Promise<HttpResponseInit> {
  if (!requireContentManager(request)) return { status: 403, body: 'Forbidden' }

  try {
    const formData = await request.formData()
    const file = formData.get('file') as File | null
    const title = formData.get('title') as string
    const description = (formData.get('description') as string) ?? ''
    const category = formData.get('category') as string
    const featured = formData.get('featured') === 'true'

    if (!file || !title || !category) return { status: 400, body: 'Missing required fields: file, title, category' }

    const ext = file.name.split('.').pop()?.toLowerCase() ?? ''
    const resourceType = ALLOWED_TYPES[ext]
    if (!resourceType) return { status: 400, body: `Unsupported file type: .${ext}` }
    if (file.size > MAX_SIZE) return { status: 400, body: 'File exceeds 100 MB limit' }

    const id = uuidv4()
    const blobName = `${id}.${ext}`
    const buffer = Buffer.from(await file.arrayBuffer())
    await uploadBlob(blobName, buffer, file.type)

    const { manifest, sha } = await readManifest()
    const caller = getCallerEmail(request) ?? 'unknown'
    const resource: Resource = {
      id, title, description, category, type: resourceType,
      fileExt: ext, blobName, fileSize: file.size,
      uploadDate: new Date().toISOString(), uploadedBy: caller, featured,
    }
    manifest.resources.unshift(resource)
    manifest.generatedAt = new Date().toISOString()
    await writeManifest(manifest, sha, `feat: upload resource "${title}"`)

    return { status: 201, jsonBody: { id } }
  } catch (err) {
    context.error('upload error', err)
    return { status: 500, body: 'Internal error' }
  }
}

app.http('upload', { methods: ['POST'], authLevel: 'anonymous', route: 'upload', handler: handleUpload })
```

```typescript
import { app, HttpRequest, HttpResponseInit, InvocationContext } from '@azure/functions'
import { deleteBlob } from '../shared/blobClient'
import { readManifest, writeManifest } from '../shared/githubClient'
import { requireContentManager } from '../shared/auth'

async function handleEdit(request: HttpRequest, context: InvocationContext): Promise<HttpResponseInit> {
  if (!requireContentManager(request)) return { status: 403, body: 'Forbidden' }

  const id = request.params.id

  try {
    const { manifest, sha } = await readManifest()
    const idx = manifest.resources.findIndex(r => r.id === id)
    if (idx === -1) return { status: 404, body: 'Resource not found' }

    if (request.method === 'DELETE') {
      const [removed] = manifest.resources.splice(idx, 1)
      await deleteBlob(removed.blobName)
      manifest.generatedAt = new Date().toISOString()
      await writeManifest(manifest, sha, `feat: delete resource "${removed.title}"`)
      return { status: 204 }
    }

    if (request.method === 'PUT') {
      const body = await request.json() as Partial<{ title: string; description: string; category: string; featured: boolean }>
      const resource = manifest.resources[idx]
      if (body.title !== undefined) resource.title = body.title
      if (body.description !== undefined) resource.description = body.description
      if (body.category !== undefined) resource.category = body.category
      if (body.featured !== undefined) resource.featured = body.featured
      manifest.generatedAt = new Date().toISOString()
      await writeManifest(manifest, sha, `feat: update resource "${resource.title}"`)
      return { status: 200, jsonBody: resource }
    }

    return { status: 405, body: 'Method not allowed' }
  } catch (err) {
    context.error('resourceEdit error', err)
    return { status: 500, body: 'Internal error' }
  }
}

app.http('resourceEdit', {
  methods: ['PUT', 'DELETE'],
  authLevel: 'anonymous',
  route: 'resource/{id}',
  handler: handleEdit,
})
```

```typescript
import { useState, DragEvent } from 'react'
import { useNavigate } from 'react-router-dom'
import './UploadPage.css'
import { useManifest } from '../../context/ManifestContext'
import { useToast } from '../../context/ToastContext'

const ALLOWED_EXTENSIONS = ['pdf','docx','pptx','xlsx','mp4','webm','jpg','jpeg','png','gif','svg']

export default function UploadPage() {
  const { state: { manifest }, refresh } = useManifest()
  const showToast = useToast()
  const navigate = useNavigate()
  const [file, setFile] = useState<File | null>(null)
  const [dragOver, setDragOver] = useState(false)
  const [title, setTitle] = useState('')
  const [description, setDescription] = useState('')
  const [category, setCategory] = useState('')
  const [featured, setFeatured] = useState(false)
  const [uploading, setUploading] = useState(false)

  const categories = manifest?.categories.map(c => c.name) ?? []

  function handleDrop(e: DragEvent) {
    e.preventDefault(); setDragOver(false)
    const f = e.dataTransfer.files[0]
    if (f) acceptFile(f)
  }

  function acceptFile(f: File) {
    const ext = f.name.split('.').pop()?.toLowerCase() ?? ''
    if (!ALLOWED_EXTENSIONS.includes(ext)) {
      showToast(`Unsupported file type: .${ext}`, 'error'); return
    }
    if (f.size > 100 * 1024 * 1024) {
      showToast('File exceeds the 100 MB size limit', 'error'); return
    }
    setFile(f)
    if (!title) setTitle(f.name.replace(/\.[^.]+$/, '').replace(/[-_]/g, ' '))
  }

  async function handleSubmit(e: React.FormEvent) {
    e.preventDefault()
    if (!file || !title.trim() || !category) { showToast('Please fill in all required fields', 'error'); return }

    setUploading(true)
    try {
      const form = new FormData()
      form.append('file', file)
      form.append('title', title.trim())
      form.append('description', description.trim())
      form.append('category', category)
      form.append('featured', String(featured))

      const res = await fetch('/api/upload', { method: 'POST', body: form })
      if (!res.ok) {
        const msg = await res.text()
        showToast(msg || 'Upload failed', 'error')
        return
      }
      const { id } = await res.json()
      showToast('Resource uploaded successfully', 'success')
      refresh()
      navigate(`/resource/${id}`)
    } catch {
      showToast('Upload failed — please try again', 'error')
    } finally {
      setUploading(false)
    }
  }

  return (
    <div className="upload-page">
      <h1>Upload Resource</h1>

      <div
        className={`dropzone ${file ? 'dropzone-file-selected' : ''} ${dragOver ? 'drag-over' : ''}`}
        onDragOver={e => { e.preventDefault(); setDragOver(true) }}
        onDragLeave={() => setDragOver(false)}
        onDrop={handleDrop}
        onClick={() => document.getElementById('file-input')?.click()}
      >
        <svg width="40" height="40" viewBox="0 0 40 40" fill="none">
          <path d="M20 8v16M12 16l8-8 8 8" stroke={file ? 'var(--color-success)' : 'var(--color-primary)'} strokeWidth="2" strokeLinecap="round"/>
          <path d="M8 28h24" stroke={file ? 'var(--color-success)' : '#E1DFDD'} strokeWidth="2" strokeLinecap="round"/>
        </svg>
        <p>{file ? `Selected: ${file.name}` : 'Drop a file here, or click to browse'}</p>
        <small>PDF, DOCX, PPTX, XLSX, MP4, WEBM, JPG, PNG, GIF, SVG · Max 100 MB</small>
        <input id="file-input" type="file" hidden accept={ALLOWED_EXTENSIONS.map(e => `.${e}`).join(',')}
          onChange={e => e.target.files?.[0] && acceptFile(e.target.files[0])} />
      </div>

      <form className="upload-form" onSubmit={handleSubmit}>
        <div className="form-field">
          <label htmlFor="title">Title *</label>
          <input id="title" value={title} onChange={e => setTitle(e.target.value)} required placeholder="Give this resource a clear, friendly name" />
        </div>
        <div className="form-field">
          <label htmlFor="description">Description</label>
          <textarea id="description" value={description} onChange={e => setDescription(e.target.value)} placeholder="Optional — briefly describe the content" />
        </div>
        <div className="form-field">
          <label htmlFor="category">Category *</label>
          <select id="category" value={category} onChange={e => setCategory(e.target.value)} required>
            <option value="">Select a category…</option>
            {categories.map(c => <option key={c} value={c}>{c}</option>)}
          </select>
        </div>
        <div className="form-field form-field-checkbox">
          <input id="featured" type="checkbox" checked={featured} onChange={e => setFeatured(e.target.checked)} />
          <label htmlFor="featured">Feature this resource on the homepage</label>
        </div>
        <div className="upload-actions">
          <button type="submit" className="btn-primary" disabled={uploading}>
            {uploading ? 'Uploading…' : 'Upload Resource'}
          </button>
          <button type="button" className="btn-secondary" onClick={() => navigate(-1)}>Cancel</button>
        </div>
        {uploading && <p className="upload-progress">Uploading file and updating the resource index…</p>}
      </form>
    </div>
  )
}
```

```yaml
      contents: write

          manifest.resources.sort((a, b) => new Date(b.uploadDate) - new Date(a.uploadDate));
```
